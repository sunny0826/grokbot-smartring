cmake_minimum_required(VERSION 3.20)
project(voice_timing_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MAIN "${FIRMWARE}/main")
if(NOT EXISTS "${LVGL}/lvgl.h")
    message(FATAL_ERROR "Pass -DLVGL=<managed LVGL source>")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP)
target_include_directories(lvgl PUBLIC "${LVGL}")
if(NOT TIMING_UI_SOURCE)
    set(TIMING_UI_SOURCE "${MAIN}/voice_ui.cpp")
endif()
add_executable(timing_ui ui.cpp "${TIMING_UI_SOURCE}" "${MAIN}/voice_motion.cpp"
    "${MAIN}/voice_timing.cpp" "${MAIN}/home_controls.cpp" "${MAIN}/fonts/voice_font_22.c")
target_include_directories(timing_ui PRIVATE "${MAIN}")
target_compile_options(timing_ui PRIVATE -Wall -Wextra -Werror)
target_link_libraries(timing_ui PRIVATE lvgl)
